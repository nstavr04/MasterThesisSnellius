
You add this if you want to specify a specific gpu partition
#SBATCH --partition=gpu_a100

Gia na trekso to script:
# sbatch simplejobtesting.sh

View available nodes etc:
# sinfo

View the queue
# squeue -u $USER

Check status of your job:
# sacct -j <JOBID> --format=JobID,JobName,Partition,State,ExitCode

View info on cpu:
# lscpu

View info on gpu:
# nvidia-smi

Log into gpu node for testing:
# ssh int3

# ls /sw/arch/RHEL9/EB_production/2024/software/cuDNN/9.5.0.50-CUDA-12.6.0/lib
